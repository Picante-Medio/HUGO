cmake_minimum_required(VERSION 3.30.0)
project(HUGO_superbuild LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_EXTENSIONS OFF)

include(ExternalProject)

if(MSVC)
    set(MY_BUILD_TYPE $<CONFIG>)
else()
    set(MY_BUILD_TYPE ${CMAKE_BUILD_TYPE})
endif()

set(COMMON_INSTALL_PREFIX ${CMAKE_BINARY_DIR}/Dependencies/${CMAKE_SYSTEM_PROCESSOR})

set(JOBS 8)

ExternalProject_Add(pugixml_external
    GIT_REPOSITORY "https://github.com/zeux/pugixml.git"
    GIT_TAG "v1.14"
    GIT_SHALLOW TRUE
    
    CMAKE_CACHE_ARGS
        -DCMAKE_INSTALL_PREFIX:PATH=${COMMON_INSTALL_PREFIX}
        -DCMAKE_BUILD_TYPE:STRING=${MY_BUILD_TYPE}
)

ExternalProject_Add(freetype_external
    GIT_REPOSITORY "https://gitlab.freedesktop.org/freetype/freetype.git"
    GIT_TAG "VER-2-14-1"
    GIT_SHALLOW TRUE
    
    CMAKE_CACHE_ARGS
        -DCMAKE_INSTALL_PREFIX:PATH=${COMMON_INSTALL_PREFIX}
        -DCMAKE_BUILD_TYPE:STRING=${MY_BUILD_TYPE}
)

if(MSVC OR MINGW OR SKBUILD)

    ExternalProject_Add(sdl2_external
        GIT_REPOSITORY "https://github.com/libsdl-org/SDL.git"
        GIT_TAG "release-2.30.7"
        GIT_SHALLOW TRUE
        
        CMAKE_CACHE_ARGS
            -DCMAKE_INSTALL_PREFIX:PATH=${COMMON_INSTALL_PREFIX}
            -DCMAKE_BUILD_TYPE:STRING=${MY_BUILD_TYPE}
            -DBUILD_SHARED_LIBS:BOOL=ON
    )

endif()

set(OGRE_OPTIONS 
    -DOGRE_BUILD_MSVC_MP:BOOL=${OGRE_BUILD_MSVC_MP}
    -DOGRE_BUILD_MSVC_ZM:BOOL=${OGRE_BUILD_MSVC_ZM})

set(OGRE_OPTIONS_OFF

    OGRE_BITES_NATIVE_DIALOG
    OGRE_BITES_STATIC_PLUGINS

    OGRE_BUILD_COMPONENT_BITES
    OGRE_BUILD_COMPONENT_BULLET
    OGRE_BUILD_COMPONENT_MESHLODGENERATOR
    OGRE_BUILD_COMPONENT_OVERLAY_IMGUI
    OGRE_BUILD_COMPONENT_PAGING
    OGRE_BUILD_COMPONENT_PROPERTY
    OGRE_BUILD_COMPONENT_TERRAIN
    OGRE_BUILD_COMPONENT_VOLUME

    OGRE_BUILD_DEPENDENCIES
    # Las dependencias se buildean a mano porque si Ogre detecta esta flag compila
    # librerías que no hacen falta, como Bullet o zlib

    OGRE_BUILD_PLUGIN_ASSIMP
    OGRE_BUILD_PLUGIN_BSP
    OGRE_BUILD_PLUGIN_DOT_SCENE
    OGRE_BUILD_PLUGIN_GLSLANG
    OGRE_BUILD_PLUGIN_RSIMAGE
    OGRE_BUILD_PLUGIN_STBI

    OGRE_BUILD_RENDERSYSTEM_D3D11
    OGRE_BUILD_RENDERSYSTEM_GLES2
    OGRE_BUILD_RENDERSYSTEM_D3D9
    OGRE_BUILD_RENDERSYSTEM_TINY
    OGRE_BUILD_RENDERSYSTEM_VULKAN

    OGRE_BUILD_SAMPLES
    OGRE_BUILD_BUILD_TESTS
    OGRE_BUILD_BUILD_TOOLS

    OGRE_CONFIG_ENABLE_QUAD_BUFFER_STEREO

    OGRE_INSTALL_DEPENDENCIES
    OGRE_INSTALL_DOCS
    OGRE_INSTALL_SAMPLES
    OGRE_INSTALL_TOOLS
    OGRE_INSTALL_VSPROPS

    OGRE_NODELESS_POSITIONING

    OGRE_STATIC
)

foreach(opt IN LISTS OGRE_OPTIONS_OFF)
    list(APPEND OGRE_OPTIONS -D${opt}:BOOL=OFF)
endforeach()

set(OGRE_OPTIONS_ON

    OGRE_BUILD_COMPONENT_OVERLAY
    #OGRE_BUILD_COMPONENT_OVERLAY_IMGUI
    OGRE_BUILD_COMPONENT_RTSHADERSYSTEM

    OGRE_BUILD_PLUGIN_OCTREE
    OGRE_BUILD_PLUGIN_PCZ
    OGRE_BUILD_PLUGIN_PFX

    OGRE_BUILD_RENDERSYSTEM_GL3PLUS
    OGRE_BUILD_RTSHADERSYSTEM_SHADERS

    OGRE_CONFIG_FILESYSTEM_UNICODE

    OGRE_COPY_DEPENDENCIES
    OGRE_ENABLE_PRECOMPILED_HEADERS
    OGRE_INSTALL_PDB

    OGRE_PROJECT_FOLDERS
)

foreach(opt IN LISTS OGRE_OPTIONS_ON)
    list(APPEND OGRE_OPTIONS -D${opt}:BOOL=ON)
endforeach()

ExternalProject_Add(ogre_external
    GIT_REPOSITORY "https://github.com/OGRECave/ogre.git"
    GIT_TAG "v14.5.2"
    GIT_SHALLOW TRUE

    DEPENDS sdl2_external pugixml_external freetype_external

    CMAKE_CACHE_ARGS
        -DOGRE_DEPENDENCIES_DIR:PATH=${COMMON_INSTALL_PREFIX}
        -DCMAKE_INSTALL_PREFIX:PATH=${COMMON_INSTALL_PREFIX}
        -DCMAKE_BUILD_TYPE:STRING=${MY_BUILD_TYPE}
        ${OGRE_OPTIONS}
)

ExternalProject_Add(hugo_external
    SOURCE_DIR ${CMAKE_CURRENT_SOURCE_DIR}/src
    BINARY_DIR ${CMAKE_CURRENT_BINARY_DIR}/hugo-build
    
    DEPENDS sdl2_external pugixml_external freetype_external ogre_external
    
    CMAKE_CACHE_ARGS
        -DCMAKE_PREFIX_PATH:PATH=${COMMON_INSTALL_PREFIX}
        -DCMAKE_BUILD_TYPE:STRING=${MY_BUILD_TYPE}
        -DCMAKE_CXX_FLAGS_DEBUG:STRING=${CMAKE_CXX_FLAGS_DEBUG}
        -DCMAKE_CXX_FLAGS_RELEASE:STRING=${CMAKE_CXX_FLAGS_RELEASE}
        -DENGINE_BIN_DIR:PATH=${CMAKE_SOURCE_DIR}/bin

    INSTALL_COMMAND ""
)
